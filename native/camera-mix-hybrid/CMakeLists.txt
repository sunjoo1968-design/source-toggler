cmake_minimum_required(VERSION 3.28)
project(camera-mix-hybrid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/camera-mix-hybrid.lua" SCRIPT)
string(REGEX MATCH "local VERSION, AUTHOR = '([^']+)'" VERSION_MATCH "${SCRIPT}")
set(HYBRID_VERSION "${CMAKE_MATCH_1}")
if(NOT HYBRID_VERSION)
  message(FATAL_ERROR "Hybrid Lua version not found")
endif()
set(OBS_SOURCE_DIR "" CACHE PATH "OBS source SDK")
set(OBS_IMPORT_DIR "" CACHE PATH "OBS import libraries")
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file("${OBS_SOURCE_DIR}/libobs/obsconfig.h.in" "${CMAKE_BINARY_DIR}/generated/obsconfig.h")
add_library(camera-mix-hybrid MODULE output.cpp)
target_include_directories(camera-mix-hybrid SYSTEM PRIVATE "${OBS_SOURCE_DIR}/libobs" "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(camera-mix-hybrid PRIVATE "${OBS_IMPORT_DIR}/obs.lib")
target_compile_definitions(camera-mix-hybrid PRIVATE HYBRID_VERSION="${HYBRID_VERSION}" WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(camera-mix-hybrid PRIVATE /W4 /utf-8 /permissive-)
set_target_properties(camera-mix-hybrid PROPERTIES PREFIX "")
