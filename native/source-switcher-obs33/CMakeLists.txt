cmake_minimum_required(VERSION 3.28)
project(source-switcher LANGUAGES C)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/buildspec.json" SPEC)
string(JSON SWITCHER_VERSION GET "${SPEC}" version)
set(OBS_SOURCE_DIR "" CACHE PATH "OBS33 source SDK")
set(OBS_IMPORT_DIR "" CACHE PATH "OBS33 import libraries")
set(OBS_RELEASE_CANDIDATE 1)
set(OBS_BETA 0)
configure_file("${OBS_SOURCE_DIR}/libobs/obsconfig.h.in" "${CMAKE_BINARY_DIR}/generated/obsconfig.h")
configure_file(version.h.in "${CMAKE_BINARY_DIR}/generated/version.h" @ONLY)
add_library(source-switcher MODULE source-switcher.c)
target_include_directories(source-switcher PRIVATE "${OBS_SOURCE_DIR}/libobs" "${OBS_SOURCE_DIR}/frontend/api" "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(source-switcher PRIVATE "${OBS_IMPORT_DIR}/obs.lib" "${OBS_IMPORT_DIR}/obs-frontend-api.lib")
target_compile_options(source-switcher PRIVATE /utf-8 /W4)
set_target_properties(source-switcher PROPERTIES PREFIX "")
